SQL PostgreSQL
CASE WHEN ((COALESCE("income", 0::float8) > 30000::float8)
      AND (COALESCE("age", 0::float8) < 60::float8))
THEN 'ผ่าน' ELSE 'ไม่ผ่าน' END
JS JavaScript / TypeScript
const f = compile(formula, { from: "excel", to: "js",
  columnMap: { B: "income", C: "age" } });
f.evaluate({ income: 40000, age: 30 }); // "ผ่าน"
